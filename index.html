async function checkIMEI() { const imei = document.getElementById('imeiInput').value.trim(); if (!imei) { showResult("⚠️ تکایە سەرەتا ژمارەی IMEI یان سەریال بنووسە یان سکانی بکە!", "error"); return; } if (imei.length < 8) { showResult("⚠️ ژمارەی IMEI یان سەریالەکە کورتە، تکایە دڵنیابەوە.", "error"); return; } showResult(`⏳ سەرقاڵی پشکنینی ژمارە:
${imei}
کەمێک چاوەڕێ بە...`, "loading"); try { // ناردنی داواکاری بۆ API فەرمی const response = await fetch(`https://alpha.imeicheck.com/api/modelBrandName?imei=${encodeURIComponent(imei)}&format=json`); if (!response.ok) { throw new Error('Network response was not ok'); } const data = await response.json(); // پشکنین بزانین زانیاری دروست گەڕاوەتەوە if (data && (data.brand || data.model || data.name)) { let htmlContent = `✅ ئەنجامی پشکنین:
`; htmlContent += `• کۆمپانیا: ${data.brand || 'نادیار'}
`; htmlContent += `• مۆدێل: ${data.model || data.name || 'نادیار'}
`; htmlContent += `• ژمارە: ${imei}`; showResult(htmlContent, 'success'); } else { showResult("❌ هیچ زانیارییەک بۆ ئەم ژمارەیە نەدۆزراوەوە. دڵنیابەوە کە ژمارەی IMEI یە نەک تەنها ژمارەی زنجیرەیی (Serial).", "error"); } } catch (error) { // ئەگەر ڕێگری CORS هەبوو لە وێبگەڕەکەدا showResult("⚠️ کێشە لە پەیوەندیکردن بە سێرڤەری پشکنین. ڕەنگە APIـیەکە ڕێگری لە داواکاری ڕاستەوخۆی وێبگەڕ بکات.", "error"); } }
